-- Create schema
CREATE SCHEMA IF NOT EXISTS payment;

-- Set search path
SET search_path TO payment, public;

-- Clean up existing objects in reverse dependency order
DROP TABLE IF EXISTS payment.payments CASCADE;
DROP TABLE IF EXISTS payment.users CASCADE;
DROP TYPE IF EXISTS payment.user_role CASCADE;
DROP TYPE IF EXISTS payment.payment_status CASCADE;
DROP TYPE IF EXISTS payment.transaction_type CASCADE;

-- Create ENUM types
CREATE TYPE payment.user_role AS ENUM (
    'ADMIN',
    'USER'
);

CREATE TYPE payment.payment_status AS ENUM (
    'FAILED',
    'PENDING',
    'SUCCESS'
);

CREATE TYPE payment.transaction_type AS ENUM (
    'DOMESTIC',
    'INTERNATIONAL',
    'PEER',
    'REMITANCE',
    'WALLET'
);

-- =========================================================================
-- 1. USERS TABLE
-- =========================================================================
CREATE TABLE payment.users (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    password VARCHAR(255) NOT NULL,
    role payment.user_role NOT NULL,
    username VARCHAR(255) NOT NULL,
    CONSTRAINT UKr43af9ap4edm43mmtq01oddj6 UNIQUE (username)
);

-- =========================================================================
-- 2. PAYMENTS TABLE
-- =========================================================================
CREATE TABLE payment.payments (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    amount NUMERIC(38, 2) DEFAULT NULL,
    created_date TIMESTAMP(6) WITHOUT TIME ZONE DEFAULT NULL,
    fee INTEGER NOT NULL,
    status payment.payment_status NOT NULL,
    transaction_type payment.transaction_type DEFAULT NULL,
    user_id BIGINT DEFAULT NULL,
    recipient_id BIGINT NOT NULL,
    CONSTRAINT FKj94hgy9v5fw1munb90tar2eje FOREIGN KEY (user_id) REFERENCES payment.users (id)
);

-- Create index for user_id lookup
CREATE INDEX idx_payments_user_id ON payment.payments (user_id);
